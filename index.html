Live API Data Generator

Click the button to load a live fact from the web!

// 4. Live API Fetch Feature const fetchBtn = document.querySelector('#fetch-btn'); const factDisplay = document.querySelector('#fact-display'); fetchBtn.addEventListener('click', async () => { factDisplay.textContent = 'Loading a fresh fact from the web... 🌐'; try { // We use a free public API endpoint for random useful facts const response = await fetch('https://uselessfacts.jsph.pl/random.json?language=en'); const data = await response.json(); // Display the fetched fact on the page factDisplay.textContent = `"${data.text}"`; } catch (error) { factDisplay.textContent = 'Oops! Could not load data. Check your internet connection.'; console.error('Fetch error:', error); } });
